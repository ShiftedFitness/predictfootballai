# TeleStats — https://telestats.net
#
# There was no robots.txt at all before this; the URL returned Netlify's 404
# page. Nothing was being blocked, but nothing was being pointed at either —
# no sitemap, no signal about what matters.
#
# The default is permissive on purpose. Team pages, game pages and the data
# pages are the whole point of the site being findable. Only admin screens,
# auth flows and the offline shell are held back, because none of them is a
# useful search result and two of them are private.

User-agent: *
Allow: /

# Fives admin and auth. Not secret, but not search results either.
Disallow: /predict/admin.html
Disallow: /predict/admin_predictions.html
Disallow: /predict/login.html
Disallow: /predict/PicksCheck.html
Disallow: /account/

# The service worker's offline fallback. It is a placeholder, not a page.
Disallow: /offline.html

# Superseded copies of the games, kept for reference. They have no canonical
# and would compete with the live /games/* pages for the same queries.
Disallow: /archive/

# A player's own profile. Not secret, but not a search result.
Disallow: /profile/

# Goal Recreator is a game to play, not a page to find.
Disallow: /goals/

# Session and challenge URLs carry a player's own result and should never
# compete with the real pages for the same query.
Disallow: /*?challenge=
Disallow: /*?session=

# A preconfigured game is a real destination for a person following a link
# from a team page, but it is the TEAM page that deserves to rank — otherwise
# hundreds of near-identical game URLs compete with the pages that carry the
# actual content.
Disallow: /games/*?scope=

# AI search crawlers, named explicitly because TeleStats answers
# database-backed factual questions and being citable is the point.
#
# EVERY RULE IS REPEATED IN FULL INSIDE EACH BLOCK, AND MUST BE.
#
# A crawler obeys only the most specific group that names it and ignores every
# other group — including `User-agent: *`. So a block containing just
# "Allow: /" does not mean "welcome, subject to the rules above"; it means
# "welcome, and none of the rules above apply to you". The first version of
# this file did exactly that, which made these six crawlers the only ones
# permitted into /account/, the Fives admin screens and the 1,500 near-
# duplicate ?scope= game URLs deliberately kept out of the index.
#
# If a Disallow is added to the wildcard block, add it here too.
